// PR #12 review fixes: browser checks on the scratch clone (port 3183).
import { chromium } from "playwright";

const BASE = process.env.BASE ?? "http://127.0.0.1:3183";
const b = await chromium.launch();
const out = {};
const watchErrors = (p, errs) => {
  p.on("pageerror", (e) => errs.push(e.message.slice(0, 100)));
  p.on("console", (m) => { if (m.type() === "error") errs.push(m.text().slice(0, 100)); });
};

// A. Go-live that lands under the stinger gets its burst once the stinger ends.
{
  const ctx = await b.newContext({ viewport: { width: 1920, height: 1080 } });
  const p = await ctx.newPage();
  const errs = [];
  watchErrors(p, errs);
  await p.addInitScript(() => {
    window.__log = [];
    const t0 = performance.now();
    const tick = () => {
      const fx = window.__fx;
      const stinger = !!document.querySelector('[data-testid="tv-intro-stinger"]');
      const live = fx ? fx.liveCount() : -1;
      const last = window.__log[window.__log.length - 1];
      if (!last || last.stinger !== stinger || last.live !== live) window.__log.push({ t: Math.round(performance.now() - t0), stinger, live });
      requestAnimationFrame(tick);
    };
    requestAnimationFrame(tick);
  });
  await p.goto(`${BASE}/tv/fx-baseline/golive?intro=1`, { waitUntil: "load" });
  await p.waitForTimeout(9000);
  const log = await p.evaluate(() => window.__log);
  const end = log.find((r) => !r.stinger && r.t > 100);
  const maxAfter = Math.max(...log.filter((r) => end && r.t >= end.t).map((r) => r.live));
  out.goLive = { stingerEndMs: end?.t, maxLiveAfterStinger: maxAfter, burstFired: maxAfter > 1, log: log.slice(0, 30), errs };
  await ctx.close();
}

// B. Slow logo (1.2 s): the stinger still plays to its end (≈ 1.2 + 4.3 s), not cut at 5 s.
{
  const ctx = await b.newContext({ viewport: { width: 1920, height: 1080 } });
  const p = await ctx.newPage();
  const errs = [];
  watchErrors(p, errs);
  await p.route("**/brand/web/htc-logo-1254.webp", async (r) => {
    await new Promise((res) => setTimeout(res, 1200));
    await r.continue();
  });
  const t0 = Date.now();
  await p.goto(`${BASE}/tv/fx-baseline/idle-2?intro=1`, { waitUntil: "commit" });
  await p.waitForSelector('[data-testid="tv-intro-stinger"]', { state: "detached", timeout: 12000 });
  out.slowLogo = { stingerMs: Date.now() - t0, errs };
  await ctx.close();
}

// C. No JS: the navy cover fades itself after ~7 s.
{
  const ctx = await b.newContext({ viewport: { width: 1920, height: 1080 }, javaScriptEnabled: false });
  const p = await ctx.newPage();
  await p.goto(`${BASE}/tv/fx-baseline/idle-2?intro=1`, { waitUntil: "load" });
  const style = () => p.$eval('[data-testid="tv-intro-stinger"]', (el) => {
    const cs = getComputedStyle(el);
    return { opacity: cs.opacity, visibility: cs.visibility, animation: cs.animationName };
  });
  const at1 = await style();
  await p.waitForTimeout(8000);
  out.noJs = { at1s: at1, at8s: await style() };
  await p.screenshot({ path: "nojs-8s.jpg", type: "jpeg", quality: 60 });
  await ctx.close();
}

// D. Title margins, E. reduced motion with intro=0 then motion back, F. resize.
{
  const ctx = await b.newContext({ viewport: { width: 1920, height: 1080 }, reducedMotion: "reduce" });
  const p = await ctx.newPage();
  const errs = [];
  watchErrors(p, errs);
  await p.goto(`${BASE}/tv/fx-baseline/lt-intermission`, { waitUntil: "load" });
  await p.waitForTimeout(2000);
  out.title = await p.$eval("h1", (el) => {
    const cs = getComputedStyle(el);
    return { marginTop: cs.marginTop, marginBottom: cs.marginBottom, paddingTop: cs.paddingTop };
  });
  const st = () => p.evaluate(() => ({ tier: window.__fx.tier(), live: window.__fx.liveCount() }));
  const reduced = await st();
  await p.emulateMedia({ reducedMotion: "no-preference" });
  await p.waitForTimeout(500);
  const motionBack = await st();
  await p.setViewportSize({ width: 1280, height: 720 });
  await p.waitForTimeout(300);
  await p.setViewportSize({ width: 1920, height: 1080 });
  await p.waitForTimeout(500);
  const afterResize = await st();
  await p.screenshot({ path: "lt-intermission.jpg", type: "jpeg", quality: 60 });
  out.ambient = { reduced, motionBack, afterResize, errs };
  await ctx.close();
}

console.log(JSON.stringify(out, null, 1));
await b.close();
