import { chromium } from "playwright";
const BASE = "http://127.0.0.1:3183";
const b = await chromium.launch();
for (const pollAt of [3000, 4800]) {
  const ctx = await b.newContext({ viewport: { width: 1920, height: 1080 } });
  const p = await ctx.newPage();
  const errs = [];
  p.on("pageerror", (e) => errs.push(e.message.slice(0, 100)));
  p.on("console", (m) => { if (m.type() === "error") errs.push(m.text().slice(0, 100)); });
  await p.addInitScript((pollAt) => {
    const st = window.setTimeout.bind(window);
    window.setTimeout = (fn, ms, ...a) => st(fn, ms === 20000 ? pollAt : ms, ...a);
    window.__log = [];
    const t0 = performance.now();
    const tick = () => {
      const fx = window.__fx;
      const stinger = !!document.querySelector('[data-testid="tv-intro-stinger"]');
      const takeover = /live/i.test(document.body?.innerText ?? "") && !!document.querySelector('[class*="z-"]') ? document.body.innerText.includes("LIVE") || document.body.innerText.includes("live") : false;
      const live = fx ? fx.liveCount() : -1;
      const last = window.__log[window.__log.length - 1];
      if (!last || last.stinger !== stinger || last.live !== live) window.__log.push({ t: Math.round(performance.now() - t0), stinger, live });
      requestAnimationFrame(tick);
    };
    requestAnimationFrame(tick);
  }, pollAt);
  await p.goto(`${BASE}/tv/fx-baseline/golive?intro=1`, { waitUntil: "load" });
  await p.waitForTimeout(9000);
  const log = await p.evaluate(() => window.__log);
  const end = log.find((r) => !r.stinger && r.t > 100);
  const maxAfter = Math.max(...log.filter((r) => end && r.t >= end.t).map((r) => r.live));
  console.log(JSON.stringify({ pollAt, stingerEndMs: end?.t, maxLiveAfterStinger: maxAfter, burstFired: maxAfter > 1, log, errs }));
  await ctx.close();
}
await b.close();
