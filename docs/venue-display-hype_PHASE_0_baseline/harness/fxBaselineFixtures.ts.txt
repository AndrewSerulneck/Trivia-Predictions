export const FX_BASELINE_FIXTURES = [
  "idle-0", "idle-1", "idle-2", "lt-question", "lt-reveal", "lt-intermission", "lt-final",
  "cb-round", "cb-intermission", "cb-results", "squares", "bracket", "golive",
] as const;

/** `golive`: the page starts idle; the state route answers with a Live Trivia question (first poll flips it). */
export const fxBaselineState = (name: string, nowMs: number, forPoll: boolean): VenueScreenState | null => {
  switch (name) {
    case "idle-0": return idle(nowMs, 0);
    case "idle-1": return idle(nowMs, 1);
    case "idle-2": return idle(nowMs, 2);
    case "lt-question": return lt(nowMs, "question");
    case "lt-reveal": return lt(nowMs, "reveal");
    case "lt-intermission": return lt(nowMs, "intermission");
    case "lt-final": return lt(nowMs, "final");
    case "cb-round": return cb(nowMs, "round");
    case "cb-intermission": return cb(nowMs, "intermission");
    case "cb-results": return cb(nowMs, "results");
    case "squares": return squares(nowMs);
    case "bracket": return bracket(nowMs);
    case "golive": return forPoll ? lt(nowMs, "question") : idle(nowMs, 2);
    default: return null;
  }
};
