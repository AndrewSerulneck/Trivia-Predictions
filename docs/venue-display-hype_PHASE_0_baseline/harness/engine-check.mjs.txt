import { chromium } from "playwright";
const OUT = process.argv[2];
const URL = "http://127.0.0.1:3157/tv/fx-harness";
const browser = await chromium.launch();
const errors = [];
const report = {};
const canvasState = (page) => page.evaluate(() => { const c = document.querySelector('[data-testid="fx-canvas"]'); return { vis: c.style.visibility, w: c.width, h: c.height, cssW: c.style.width }; });
const fx = (page, expr) => page.evaluate(expr);
const frames = (page, ms) => page.evaluate((ms) => new Promise((res) => { const d = []; let last = performance.now(); const end = last + ms; const f = (n) => { d.push(n - last); last = n; if (n < end) requestAnimationFrame(f); else res(d.slice(1)); }; requestAnimationFrame(f); }), ms);
const stat = (d) => { const m = d.reduce((a, b) => a + b, 0) / d.length; const s = [...d].sort((a, b) => a - b); return { frames: d.length, meanMs: +m.toFixed(1), p95Ms: +s[Math.floor(0.95 * s.length)].toFixed(1), fps: +(1000 / m).toFixed(1) }; };

{
  const ctx = await browser.newContext({ viewport: { width: 1920, height: 1080 } });
  const page = await ctx.newPage();
  page.on("console", (m) => { if (m.type() === "error") errors.push(m.text()); });
  page.on("pageerror", (e) => errors.push(e.message));
  await page.goto(URL, { waitUntil: "load" });
  await page.waitForFunction(() => window.__fx);
  report.atLoad = await canvasState(page);
  report.stageTransformAtLoad = await page.evaluate(() => [...document.querySelectorAll('[data-testid="brand-stage"] > div')].map((d) => getComputedStyle(d).transform));
  report.beforeArm = await fx(page, () => window.__fx.burst({ key: "k0" }));
  await page.waitForTimeout(1300);
  report.afterArmRepeatOfK0 = await fx(page, () => window.__fx.burst({ key: "k0" }));
  report.tier = await fx(page, () => window.__fx.tier());
  report.play = await fx(page, () => window.__fx.burst({ key: "k1", at: document.getElementById("anchor"), shockwave: true }));
  report.dup = await fx(page, () => window.__fx.burst({ key: "k1", at: document.getElementById("anchor"), shockwave: true }));
  await page.waitForTimeout(250);
  report.whileActive = await canvasState(page);
  await page.screenshot({ path: `${OUT}/burst-0.25s.jpg`, type: "jpeg", quality: 82 });
  await page.waitForTimeout(300);
  await page.screenshot({ path: `${OUT}/burst-0.55s.jpg`, type: "jpeg", quality: 82 });
  report.shower = await fx(page, () => window.__fx.shower({ key: "s1", kind: "mixed", colors: ["#22d3ee", "#a78bfa", "#fbbf24"] }));
  await page.waitForTimeout(1200);
  await page.screenshot({ path: `${OUT}/shower-1.2s.jpg`, type: "jpeg", quality: 82 });
  await page.waitForTimeout(3000);
  report.afterAllDone = await canvasState(page);
  await page.evaluate(() => { window.__stop = window.__fx.twinkles({ count: 30 }); });
  await page.waitForTimeout(2500);
  await page.screenshot({ path: `${OUT}/twinkles.jpg`, type: "jpeg", quality: 82 });
  report.ambientFrames = stat(await frames(page, 2000));
  await page.evaluate(() => window.__stop());
  await page.waitForTimeout(200);
  report.afterTwinkleStop = await canvasState(page);
  report.stageTransformLater = await page.evaluate(() => [...document.querySelectorAll('[data-testid="brand-stage"] > div')].map((d) => getComputedStyle(d).transform));

  // Throttled ×6: frames during burst+shockwave+shower, and the tier afterwards.
  const cdp = await ctx.newCDPSession(page);
  report.idleFramesX6 = (await cdp.send("Emulation.setCPUThrottlingRate", { rate: 6 }), stat(await frames(page, 2000)));
  await fx(page, () => { window.__fx.burst({ key: "t1", at: document.getElementById("anchor"), shockwave: true }); window.__fx.shower({ key: "t2" }); });
  report.effectFramesX6 = stat(await frames(page, 2500));
  report.tierAfterX6 = await fx(page, () => window.__fx.tier());
  await cdp.send("Emulation.setCPUThrottlingRate", { rate: 1 });
  await page.waitForTimeout(3500);
  report.afterX6 = await canvasState(page);
  await ctx.close();
}
{
  const ctx = await browser.newContext({ viewport: { width: 1920, height: 1080 }, reducedMotion: "reduce" });
  const page = await ctx.newPage();
  page.on("pageerror", (e) => errors.push(e.message));
  await page.goto(URL, { waitUntil: "load" });
  await page.waitForFunction(() => window.__fx);
  await page.waitForTimeout(1300);
  report.reduced = { tier: await fx(page, () => window.__fx.tier()), burst: await fx(page, () => window.__fx.burst({ key: "r1" })), canvas: await canvasState(page) };
  await ctx.close();
}
{
  // 4K-ish backing cap: 1920×1080 CSS at DPR 2 → 2880 long side.
  const ctx = await browser.newContext({ viewport: { width: 1920, height: 1080 }, deviceScaleFactor: 2 });
  const page = await ctx.newPage();
  await page.goto(URL, { waitUntil: "load" });
  await page.waitForFunction(() => window.__fx);
  await page.waitForTimeout(1300);
  await fx(page, () => window.__fx.burst({ key: "hd", at: document.getElementById("anchor") }));
  await page.waitForTimeout(100);
  report.dpr2 = await canvasState(page);
  await ctx.close();
}
report.errors = errors;
console.log(JSON.stringify(report, null, 1));
await browser.close();
